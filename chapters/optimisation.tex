\section{Optimisation}
\label{cap:optimisation}

\textbf{NB:} This chapter follows \cite{pml1Book} and \cite{deisenroth2020mathematics}. For a deeper presentation, also see \cite{nesterov2018lectures} and \cite{boyd2004convex}.\\

\noindent Optimisation is central to ML: in the vast majority of cases, models are \emph{trained} by either minimising a loss function or maximising a reward function. In general, model design involves the definition of a training objective or \textbf{loss}, that is, a function that denotes \textbf{how well a model fits the data}. This objective is a function of the dataset of available observations and the chosen model, the latter usually represented by its parameters. The best model is thus chosen by optimising this objective. 

\begin{example}[Linear regression (LR)]

In the LR setting, we aim to determine the function
\begin{align}
  f \colon \R^M &\to \R\nonumber\\
  x &\mapsto f(x)=a^\top x + b,\quad a\in\R^M,b\in\R
   \label{eq:reg_lin_fn} 
\end{align}
conditional to a set of observations
\begin{equation}
	\datos=\{(x_i,y_i)\}_{i=1}^N\subset \R^M \times \R.
	\label{eq:training_set}
\end{equation}
 Using least squares, the function $f$ is chosen via minimisation of the sum of the square differences between observations $\{y_i\}_{i=1}^N$ and predictions $\{f(x_i)\}_{i=1}^N$. That is, we aim to minimise the loss:
\begin{equation}
	J(\datos,f) = \sum_{i=1}^N(y_i-f(x_i))^2 = \sum_{i=1}^N(y_i-a^
	\top x_i - b)^2.
	\label{eq:least_squares_cost}
\end{equation}
We show an example of a linear model learnt from data in Figure \ref{fig:lr_crickets}.

    \caption{Example of a linear regression model minimising the least squares loss.}
    \label{fig:lr_crickets} 
    \end{minipage}
\end{center}
\end{example}


\begin{example}[Logistic regression]

Here, we aim to determine the function
\begin{align}
  f \colon \R^M &\to \R\nonumber\\
  x &\mapsto f(x)= \frac{1}{1 + e^{-\theta^\top x+b}},\quad \theta\in\R^M, b\in\R,
   \label{eq:log_reg_fn} 
\end{align}
conditional to the  observations
\begin{equation}
	\datos=\{(x_i,c_i)\}_{i=1}^N\subset \R^M \times \{0,1\}.
	\label{eq:training_set_classif}
\end{equation}
 The standard loss function for the classification problem is the cross entropy, given by:
\begin{align}
	J(\datos,f) &= -\frac{1}{N} \sum_{i=1}^N \left(c_i\log f(x_i) + (1-c_i)\log(1-f(x_i))\right)\\
				&=  \frac{1}{N} \sum_{i=1}^N \left( \log(1+e^{-\theta^\top x_i+b}) -(1-c_i)(-\theta^\top x_i+b)  \right).
	\label{eq:x_entropy_cost}
\end{align}
Figure \ref{fig:log_reg_example} shows an example of a binary classification task minimising the cross entropy to separate data generated from two Gaussian distributions.

\end{center}
\end{example}


\begin{example}[Clustering ($K$-means)]

Given a set of observations
\begin{equation}
	\datos=\{x_i\}_{i=1}^N\subset \R^M,
	\label{eq:training_set_clustering}
\end{equation}
we aim to find cluster centres (or prototypes) $\mu_1,\mu_2,\ldots, \mu_K$ and \emph{assignment variables} $\{r_{ik}\}_{i,k=1}^{N,K}$, to minimise the following loss 
\begin{align}
	J(\datos,f) = \sum_{i=1}^N\sum_{k=1}^K r_{ik} ||x_i-\mu_k||^2.
	\label{eq:clustering_cost} 
\end{align}

\subsection{Terminology} % (fold)
\label{sub:opt_terminology}

We write a constrained optimisation problem as
\begin{equation}
    \begin{aligned}
        &\min_{x\in\cX} f(x)\\
        &\text{subject to } g_i(x)\leq 0,\quad i=1,\ldots,I,\\
        &\phantom{\text{subject to }} h_j(x)=0,\quad j=1,\ldots,J.
    \end{aligned}
\end{equation}
Its components are:

\begin{itemize}
    \item \textbf{Objective function:} The function $f:\cX\to\R$ assigns a value to each candidate $x$. We seek to minimise this value.
    \item \textbf{Constraints:} The inequalities $g_i(x)\leq 0$ and equalities $h_j(x)=0$ specify which choices of $x$ are allowed.
    \item \textbf{Feasible region:} The set of all points satisfying the constraints is
    \begin{equation}
        C=\bigl\{x\in\cX\mid g_i(x)\leq 0\ \forall i=1,\ldots,I,\quad
        h_j(x)=0\ \forall j=1,\ldots,J\bigr\}.
    \end{equation}
    For an unconstrained problem, $C=\cX$.
    \item \textbf{Optimisation variable and minimiser:} The variable $x$ is the quantity we choose. A global minimiser $x_\star$, when one exists, satisfies
    \begin{equation}
        x_\star\in\argmin_{x\in C} f(x),
    \end{equation}
    where $C$ is the feasible region defined below. The operator $\argmin$ denotes the \emph{set} of minimisers; the minimum value is $f(x_\star)$. A minimiser need not exist, and it need not be unique.
    \item \textbf{Local and global minimisers:} Let $x_\star\in C$. It is a local minimiser if it minimises $f$ among nearby \emph{feasible} points:
    \begin{equation}
        \exists\,\delta>0\quad\text{such that}\quad
        f(x_\star)\leq f(x)\quad
        \forall x\in C\text{ with }\norm{x-x_\star}<\delta.
    \end{equation}
    It is a global minimiser if it minimises $f$ over the entire feasible region:
    \begin{equation}
        f(x_\star)\leq f(x)\quad\forall x\in C.
    \end{equation}
    Every global minimiser is also a local minimiser.
\end{itemize}

\begin{example}[Unique and non-unique global minimisers]
Consider the functions $f,g\colon\mathbb{R}^2\to\mathbb{R}$ defined by
\begin{equation*}
    f(x,y) = (x-1)^2+(y+2)^2
\end{equation*}
and
\begin{equation*}
    g(x,y) = (x^2-1)^2+(y^2-1)^2.
\end{equation*}
The functions $f$ and $g$ are plotted in the left and right panels of Figure~\ref{fig:example_local_global}, respectively.
Both functions are sums of squares and attain their global minimum of $0$ precisely where both squared terms vanish.
Thus $f$ has the unique global minimiser $(1,-2)$, whereas the set of global minimisers of $g$ is $\{-1,1\}\times\{-1,1\}$.

\subsection{Continuous unconstrained optimisation} % (fold)
\label{sec:continuous_optimisation}

\begin{remark}[Optimisation variable]
In this section, the optimisation variable is the model parameter vector $\theta$.
\end{remark}

We now consider unconstrained optimisation on an open set $\Theta\subseteq\R^d$. We seek a minimiser of the form

\begin{equation}
	\theta_\star \in \argmin_{\theta\in\Theta} L(\theta).
\end{equation}
If $\theta_\star\in\Theta$ satisfies the above, then
\begin{equation}
	\forall \theta\in\Theta,~ L(\theta_\star) \leq L(\theta),
\end{equation}
meaning that it is a \textbf{global minimiser}. Finding one may be difficult, so we also consider local minimisers: points $\theta_\star\in\Theta$ such that 
\begin{equation}
\exists\, \delta > 0  \;\; 
\text{s.t.} \;\; \forall\, \theta \in \Theta \;\; 
\|\theta - \theta_\star\| < \delta 
\;\Rightarrow\;
L(\theta_\star) \le L(\theta).
\end{equation}

We now review the \textbf{optimality conditions}.

\begin{assumption}
The loss function $L:\Theta\to\R$ is twice continuously differentiable on the open set $\Theta$.
\end{assumption}

Write $g(\theta)=\nabla L(\theta)$ for the gradient and $H(\theta)=\nabla^2 L(\theta)$ for the Hessian. A point where $g(\theta)=0$ is called a \emph{stationary point}. The following conditions apply:

\begin{itemize}
    \item \textbf{First order necessary condition:} If $\theta_\star$ is a local minimiser, then $g(\theta_\star)=0$.
    \item \textbf{Second order necessary condition:} If $\theta_\star$ is a local minimiser, then $g(\theta_\star)=0$ and $H(\theta_\star)$ is positive semidefinite.
    \item \textbf{Second order sufficient condition:} If $g(\theta_\star)=0$ and $H(\theta_\star)$ is positive definite, then $\theta_\star$ is a \emph{strict} local minimiser: every sufficiently close distinct point has a strictly larger loss.
\end{itemize}

\begin{remark}
Positive definiteness is sufficient, but not necessary, even for a strict local minimum: $L(\theta)=\theta^4$ has a strict global minimum at $0$, yet $L''(0)=0$. A positive semidefinite Hessian at a stationary point is not sufficient either: for $L(\theta)=\theta^3$, both $L'(0)$ and $L''(0)$ vanish, but $0$ is not a local minimum.
\end{remark}

\begin{example}[Different stationary points]

For a fixed parameter $p\in\R$, consider the function
\begin{align}
  f \colon \R^2 &\to \R\nonumber\\
  (x,y) &\mapsto f(x,y)=(p-1)x^2 + (p+1)y^2
   \label{eq:example_fn_stationary_points} 
\end{align}
Observe that
\begin{equation}
	\nabla f = \begin{bmatrix}   2(p-1)x \\ 2(p+1)y   \end{bmatrix},
	\label{eq:nabla_example_fn_stationary_points}
\end{equation}
so the origin is the unique stationary point when $p\neq\pm1$. Furthermore, 
\begin{equation}
	\nabla^2 f = \begin{bmatrix}   2(p-1) & 0 \\ 0 & 2(p+1)   \end{bmatrix},
	\label{eq:nabla2_example_fn_stationary_points}
\end{equation}
The eigenvalues of the Hessian are $2(p-1)$ and $2(p+1)$, giving the following cases:

\begin{itemize}
    \item $p>1$: The Hessian is positive definite, and the origin is the unique strict global minimiser.
    \item $-1<p<1$: The Hessian has one positive and one negative eigenvalue, and the origin is a \emph{saddle point}.
    \item $p<-1$: The Hessian is negative definite, and the origin is the unique strict global maximiser.
    \item $p=1$: We have $f(x,y)=2y^2$. Every point on the $x$ axis is stationary and is a global minimiser, with value $0$. None is a strict local minimiser.
    \item $p=-1$: We have $f(x,y)=-2x^2$. Every point on the $y$ axis is stationary and is a global maximiser, with value $0$. None is a strict local maximiser.
\end{itemize}
Figure \ref{fig:critical_points} illustrates the three cases with $p\neq\pm1$.

\subsection{Convex optimisation}
\label{subsec:convex_opt}


A convex optimisation problem minimises a convex objective function over a convex feasible region. Its central property is that \emph{every local minimiser is a global minimiser}; differentiability is not required for this result. We first define convexity and then develop useful conditions for differentiable functions.

\begin{definition}[Convex set]
A set $\cS\subseteq\R^d$ is convex if it contains the line segment joining any two of its points:
\begin{equation}
    \lambda x+(1-\lambda)x'\in\cS,
    \qquad\forall x,x'\in\cS,\quad\forall\lambda\in[0,1].
\end{equation}
\end{definition}
Figure \ref{fig:convex_nonconvex_sets} shows two convex and two non-convex sets.

\begin{figure}[ht]
    \centering
    \includegraphics[width=0.9\textwidth]{img/optimisation_convex_and_nonconvex_regions.pdf}
    \caption{Examples of convex and non-convex sets, highlighted in blue.}
    \label{fig:convex_nonconvex_sets}
\end{figure}

\begin{definition}[Epigraph of a function]
The epigraph of $f:\cX\subseteq\R^d\to\R$ is the set of points on or above its graph:
\begin{equation}
    \operatorname{epi}(f)=\{(x,r)\in\cX\times\R\mid f(x)\leq r\}.
\end{equation}
\end{definition}

\begin{definition}[Convex function]
A function $f:\cX\to\R$ on a convex domain $\cX\subseteq\R^d$ is convex if
\begin{equation}
    f\bigl(\lambda x+(1-\lambda)x'\bigr)
    \leq\lambda f(x)+(1-\lambda)f(x'),
    \quad\forall x,x'\in\cX,\quad\forall\lambda\in[0,1].
\end{equation}
Equivalently, $f$ is convex if its epigraph is a convex set. The function is \emph{strictly convex} if the inequality is strict whenever $x\neq x'$ and $0<\lambda<1$.
\end{definition}

\begin{example}[Convex functions (in 1D)]

The following functions are convex on the indicated domains:

\begin{itemize}
	\item $f(x)=x^2$, on $\R$.
	\item $f(x)=e^{ax}$, on $\R$, for any fixed $a\in\R$.
	\item $f(x)=-\log x$, on $(0,\infty)$.
	\item $f(x)=x^a$, on $(0,\infty)$, for $a>1$.
	\item $f(x)=|x|^a$, on $\R$, for $a\geq1$.
	\item $f(x) = x\log x,\, x>0$
\end{itemize}
Figure \ref{fig:epigraphs} shows the epigraphs for these functions.

We now review some useful results for differentiable functions.

\begin{proposition}[Non-negative derivative]
Let $f:I\to\R$ be differentiable on an open interval $I\subseteq\R$. If $f'(x)\geq0$ for every $x\in I$, then $f$ is non-decreasing on $I$.
\end{proposition}
\begin{proof}
For any $a,b\in I$ with $a<b$, the mean value theorem gives a point $c\in(a,b)$ such that
\begin{equation}
    f(b)-f(a)=f'(c)(b-a)\geq0.
\end{equation}
Thus $f(b)\geq f(a)$.
\end{proof}

\begin{proposition}[Direction of steepest ascent]
\label{prop:steepest_ascent}
Let $f:\cX\to\R$ be differentiable on an open set $\cX\subseteq\R^d$. At a point $x_0\in\cX$ where $\nabla f(x_0)\neq0$, the gradient $\nabla f(x_0)$ points in the direction of greatest local increase of $f$, to first order. For steps of a fixed small Euclidean length, the corresponding unit direction is
\begin{equation}
    u_\star=\frac{\nabla f(x_0)}{\norm{\nabla f(x_0)}}.
\end{equation}
\end{proposition}
\begin{proof}
Consider a small step $x_0+\rho u$, where $u\in\R^d$, $\norm{u}=1$, and $\rho>0$. Differentiability gives
\begin{equation}
    f(x_0+\rho u)=f(x_0)+\rho\nabla f(x_0)^\top u+o(\rho),
    \qquad\rho\to0.
\end{equation}
To maximise the increase in $f$ to first order, we therefore maximise $\nabla f(x_0)^\top u$ over unit vectors $u$. By the Cauchy--Schwarz inequality, this inner product is at most $\norm{\nabla f(x_0)}$, with equality when $u=u_\star$. Thus the greatest first-order increase is obtained by moving along the gradient. If the gradient vanishes, the first-order term is zero in every direction, and higher-order terms determine how $f$ changes.
\end{proof}

\begin{theorem}[Hessian characterisation of convexity]
\label{thm:hessian_convexity}
Let $f:\cX\to\R$ be twice continuously differentiable on an open convex set $\cX\subseteq\R^d$. Then $f$ is convex if and only if $\nabla^2 f(x)$ is positive semidefinite for every $x\in\cX$.
\end{theorem}
\begin{proof}
\textbf{One dimension.} Suppose first that $f''\geq0$ on an open interval. Then $f'$ is non-decreasing. For any two points $z,w$ in the interval, the fundamental theorem of calculus gives
\begin{equation}
    f(w)\geq f(z)+f'(z)(w-z).
    \label{eq:tanget_PD1}
\end{equation}
Indeed, if $w>z$, integrate $f'(s)\geq f'(z)$ from $z$ to $w$; if $w<z$, integrate $f'(s)\leq f'(z)$ from $w$ to $z$ and rearrange. Thus the graph lies above each tangent.

Let $a<b$, $t\in[0,1]$, and $z=(1-t)a+tb$. Apply \eqref{eq:tanget_PD1} with the \emph{same tangent point $z$} to both endpoints:
\begin{align}
    f(a)&\geq f(z)+f'(z)(a-z),\\
    f(b)&\geq f(z)+f'(z)(b-z).
\end{align}
Multiply the first inequality by $1-t$ and the second by $t$, then add. Since $(1-t)(a-z)+t(b-z)=0$, the derivative terms cancel, leaving
\begin{equation}
    f\bigl((1-t)a+tb\bigr)\leq(1-t)f(a)+tf(b).
\end{equation}
This is the convexity inequality: the graph lies below the chord joining the endpoints.

Conversely, suppose $f$ is convex. Applying convexity at the midpoint of $x-h$ and $x+h$ gives
\begin{equation}
    f(x+h)-2f(x)+f(x-h)\geq0.
\end{equation}
Divide by $h^2$ and let $h\to0$. Twice differentiability gives $f''(x)\geq0$.

\textbf{Higher dimensions.} Restrict $f$ to any line: for $x\in\cX$ and $v\in\R^d$, define $\phi(s)=f(x+sv)$ for all $s\in\R$ such that $x+sv\in\cX$. Its domain is an open interval (or all of $\R$), and the chain rule gives
\begin{equation}
    \phi''(s)=v^\top\nabla^2 f(x+sv)v.
\end{equation}
If the Hessian is positive semidefinite everywhere, every such restriction is convex by the result above. Applying this to each line segment in $\cX$ proves that $f$ is convex. Conversely, if $f$ is convex, each restriction is convex, so $\phi''(0)=v^\top\nabla^2 f(x)v\geq0$ for every $v$. This is precisely positive semidefiniteness of the Hessian.
\end{proof}

\begin{remark}[Local and global minimisers]
To see why a local minimiser of a convex function must be global, suppose $x_\star$ is local but some feasible $y$ satisfies $f(y)<f(x_\star)$. Convexity gives
\[
    f\bigl((1-t)x_\star+ty\bigr)
    \leq(1-t)f(x_\star)+tf(y)<f(x_\star),\qquad0<t<1.
\]
For sufficiently small $t$, this contradicts local minimality. A strictly convex function has at most one global minimiser, but a minimiser need not exist.
\end{remark}

\begin{example}[Using gradients and Hessians]
Consider three functions on $\R^2$.
\begin{enumerate}
    \item \textbf{A strictly convex quadratic.} For $f(x,y)=x^2+xy+y^2$,
    \[
        \nabla f=\begin{bmatrix}2x+y\\x+2y\end{bmatrix},
        \qquad\nabla^2 f=\begin{bmatrix}2&1\\1&2\end{bmatrix}.
    \]
    The Hessian has eigenvalues $1$ and $3$, so it is positive definite everywhere. The unique stationary point $(0,0)$ is the unique global minimiser, with value $0$.
    \item \textbf{A convex function with a singular Hessian at its minimum.} For $f(x,y)=x^4+y^4$,
    \[
        \nabla f=\begin{bmatrix}4x^3\\4y^3\end{bmatrix},
        \qquad\nabla^2 f=\begin{bmatrix}12x^2&0\\0&12y^2\end{bmatrix}.
    \]
    The Hessian is positive semidefinite everywhere, so $f$ is convex. Its unique global minimiser is $(0,0)$, since $x^4+y^4>0$ at every other point. The minimum is strict even though the Hessian there is zero.
    \item \textbf{A non-convex function.} For $f(x,y)=x^2-y^2$,
    \[
        \nabla f=\begin{bmatrix}2x\\-2y\end{bmatrix},
        \qquad\nabla^2 f=\begin{bmatrix}2&0\\0&-2\end{bmatrix}.
    \]
    The Hessian is indefinite, so $f$ is not convex. The origin is a stationary saddle point: $f(x,0)>0$ for $x\neq0$, whereas $f(0,y)<0$ for $y\neq0$. The function is unbounded below and has no global minimiser.
\end{enumerate}
\end{example}

\subsection{Gradient descent}
\label{sub:first_order_methods}

In general, finding a minimum by setting $\nabla f(x) = 0$ and solving for $x$ is often impractical; moreover, a stationary point need not be a minimum. For that reason, we will consider iterative methods based on gradients. The idea here is to go \emph{downhill} following the negative gradient towards a minimum (ignoring the curvature information for now). 

We will specify a starting point $x_0$ and calculate
\begin{equation}
	x_{t+1} = x_t + \eta_t d_t,
\end{equation}
where $\eta_t>0$ is a \emph{step size} and $d_t$ is a \emph{descent direction}, such as $-\nabla f(x_t)$. Here, the subindex $\cdot_t$ represents the iteration number (starting from iteration $t=0$). We iterate until convergence, that is, until the sequence $x_0,x_1,x_2,\ldots$ approaches a limiting value. In practice, we may stop when successive iterates become very similar, but this alone does not guarantee that a minimum has been found. 

Note that there are several \emph{descent directions}, that is, directions $d_t$ such that, for all sufficiently small positive step sizes, 
\begin{equation}
	f(x_t + \eta_t d_t) < f(x_t).
\end{equation}
In fact, as long as $d_t^\top\nabla f(x_t)<0$, $d_t$ is a descent direction. By Proposition~\ref{prop:steepest_ascent} applied to $-f$, whose gradient is $-\nabla f$, choosing $d_t=-\nabla f(x_t)$ gives the \emph{steepest descent direction} for the Euclidean norm, provided the gradient is nonzero.


\subsubsection{Role of the step size}
\label{ssub:stepsize}

The step size $\eta_t$ is also known as \emph{learning rate}. Furthermore,  we refer to the set $\{\eta_1,\eta_2,\ldots\}$ as the learning rate schedule. We will usually consider a constant learning rate, that is, $\eta_t = \eta$, $\forall t\in\N$. Though this is the simplest choice, there are some concerns to it: if $\eta$ is too large, the iteration may fail to converge; whereas if it is too small, convergence may be very slow. 


The learning rate is usually tuned based on heuristics. When implementing the rule
\begin{equation}
    x_{t+1} = x_t - \eta \nabla f(x_t),
\end{equation}
the appropriate learning rate depends on the curvature of $f$.

\begin{example}[Convergence for a parabola]
\label{ex:parabola_lr}

Let us consider the function
\begin{equation}
	J = (\theta -3)^2.
\end{equation}

For this parabola, the error $e_t=\theta_t-3$ satisfies $e_{t+1}=(1-2\eta)e_t$. This is a stable autoregressive system when $|1-2\eta|<1$, that is, $0<\eta<1$.

In Figure \ref{fig:parabola_lr} we show how the steepest descent converges/diverges for different learning rates.

\subsubsection{Momentum}
\label{ssub:momentum}

In higher dimensions, we want to move faster in some directions and slower in others, depending on the value of the gradient in each coordinate. This can be achieved by: 
\begin{align}
	m_t &= \beta m_{t-1} + \nabla f(x_{t-1})\\
	x_t &= x_{t-1} - \eta_t m_t,
\end{align}
where $m_t$ accumulates past gradients, $m_0=0$, and $\beta\in[0,1)$ is a design (memory) parameter. This way, previous values of the gradient have effect on future updates: if a particular coordinate of the gradient consistently has the same sign, its contributions accumulate, whereas contributions that alternate in sign can cancel. This is particularly useful when the evaluation of the gradient is noisy.

\label{ex:parabola_momentum}

\subsubsection{Newton method}
\label{ssub:newton}
For comparison, Newton's method uses second-order information:
\begin{equation}
	x_{t+1} = x_t -\eta_t H_t^{-1}\nabla f(x_t),	
\end{equation}
where recall that $H_t = \nabla^2 f(x_t)$ denotes the Hessian of $f$ at $x_t$, assumed invertible. This update follows from considering the second order approximation of the loss function around the current point, that is: 

\begin{equation}
	f(x) \simeq f(x_t) + (\nabla f(x_t))^\top(x-x_t) + \frac{1}{2} (x-x_t)^\top H_t  (x-x_t).
\end{equation}
If $H_t$ is positive definite, the minimum of this quadratic approximation is given by 
\begin{equation}
	\tilde{x}_\star = x_t - H^{-1}_t\nabla f(x_t),
\end{equation}
where $\tilde{x}_\star$ minimises the approximation, rather than necessarily $f$. Taking $\eta_t=1$ gives this full Newton step; a smaller positive learning rate damps the step, which can help convergence. 

\begin{example}[Newton's method on a parabola]

Apply Newton's method to the objective $J(\theta)=(\theta-3)^2$ used in Example~\ref{ex:parabola_momentum}.

As shown in Figure \ref{fig:newton_method}, this method converges in one step with $\eta=1$ (for a quadratic function with a positive definite Hessian). That is, the update corresponds to the closed form solution of the minimisation problem.

\subsection{Stochastic gradient descent}
\label{sub:SGD}

\begin{wrapfigure}{r}{0.48\textwidth}
    \centering
    \resizebox{0.96\linewidth}{!}{\input{img/optimisation_sgd_foundation}}
\end{wrapfigure}

We now consider stochastic optimisation, where
\begin{equation}
	L(\theta) = \E_{q(x)}L(\theta,x),
\end{equation}
that is, when the loss function is random and we aim to minimise its expected value.  For instance, in linear regression we have $L(\theta)=\E_{q(y|x)}(y-\theta^\top x)$.

In practice, we are unable to compute this expectation since the law $q(x)$ is unknown. However, since we usually have samples of $q$, we can do a sample approximation of the expectation. In fact, we will consider
\begin{equation}
	L(\theta) = \frac{1}{N}\sum_{i=1}^nL(\theta,x_i).
\end{equation}
In the linear regression example, this would be $L(\theta) = \frac{1}{N}\sum_{i=1}^n(y_i-\theta^\top x_i)^2.$ 

The gradient is then also approximated using a batch of, say, $B$ samples. That is, 

\begin{equation}
	\nabla L(\theta)\simeq \frac{1}{N}\sum_{i=1}^B\nabla L(\theta,x_i).
\end{equation}

\WFclear
\begin{example}[Estimating the expected loss in linear regression]

Consider observations generated by $y_i=x_i+\epsilon_i$, where the noise terms are independent and $\epsilon_i\sim\mathcal{N}(0,\sigma^2)$. Thus $q(y\mid x)=\mathcal{N}(y;x,\sigma^2)$, and the true regression function is $f(x)=x$.

Suppose we use this predictor. Its squared error on observation $i$ is $(y_i-x_i)^2=\epsilon_i^2$: even though the regression function is correct, the noise makes the loss random. Its expected value is
\[
    \E\bigl[(y_i-x_i)^2\bigr]=\E[\epsilon_i^2]=\sigma^2.
\]
We can estimate this expectation using the average squared error over $N$ observations:
\[
    \widehat{L}_N=\frac{1}{N}\sum_{i=1}^N(y_i-x_i)^2.
\]
Figure \ref{fig:stochastic_loss} illustrates this for $\sigma=1$. The left panel shows the noisy observations and the true regression function. The right panel shows how $\widehat{L}_N$ fluctuates as observations are added, approaching the expected loss $\sigma^2=1$. Note that the curve illustrates estimation of the expected loss, not parameter  updates.

Recall that, in general, we will consider parameter updates of the form 
\begin{equation}
	\theta_{t+1} = \theta_t -M_t^{-1} g_t,
\end{equation}
where $M_t$ is an estimate of the magnitudes of the coordinates of the gradient $g_t$, as this ensures convergence (or prevents divergence). We next explore some different choices of this estimate.

\subsubsection{Adagrad}

The adaptive gradient (Adagrad) method considers the following iterative rule ($d$ denotes the coordinate):  
\begin{equation}
	\theta_{t+1,d} = \theta_{t,d} - \eta_t \frac{1}{\sqrt{s_{t,d}+\epsilon}} g_{t,d},
\end{equation}
where $s_{t,d} = \sum_{i=1}^t g_{i,d}^2$ and $\epsilon>0$. Writing $\Delta\theta_t=\theta_{t+1}-\theta_t$, the vector form is
\begin{equation}
	\Delta \theta_t = -\eta_t \frac{g_t}{\sqrt{s_t+\epsilon\mathbf{1}}},
\end{equation}
where $\mathbf{1}$ is the all-ones vector and the square root and division are applied elementwise.
The aim of Adagrad is to control the magnitude of the step size for each coordinate of the parameter independently, based on a rolling estimate using previous gradient evaluations. Notice that the computation of $s_{t,d}$ assigns the same importance to all gradient evaluations in time.


\subsubsection{RMSProp}

Adagrad might fail to represent the current gradient magnitude by not emphasising current evaluations of the gradient. This can be solved by replacing the sum in the computation of $s_{t,d}$  by a moving average. That is, 

\begin{equation}
	s_{t,d} = \beta s_{t-1,d} + (1-\beta) g_{t,d}^2,
\end{equation}
where the \emph{forgetting factor} $\beta$ is usually chosen close to 0.9. Then, the update rule also follows
\begin{equation}
	\Delta \theta_t = -\eta_t \frac{g_t}{\sqrt{s_t+\epsilon\mathbf{1}}}.
\end{equation}
 
\subsubsection{Adam}

Adam (Adaptive Moment Estimation) \cite{kingma2015adam} maintains exponential moving averages of the stochastic gradient and its elementwise square. Initialise $m_0=s_0=0$ and let $g_t$ be the stochastic gradient evaluated at $\theta_{t-1}$. For decay rates $\beta_m,\beta_s\in[0,1)$, the moment estimates are
\begin{align}
	m_t = \beta_m m_{t-1} + (1-\beta_m) g_{t}\\
	s_t = \beta_s s_{t-1} + (1-\beta_s) g^2_{t}.
\end{align}
To correct for zero initialisation, define
\begin{equation*}
    \widehat m_t = \frac{m_t}{1-\beta_m^t},
    \qquad
    \widehat s_t = \frac{s_t}{1-\beta_s^t}.
\end{equation*}
The parameter update is
\begin{equation}
	\theta_t = \theta_{t-1} - \eta_t \frac{\widehat m_t}{\sqrt{\widehat s_t} + \epsilon\mathbf{1}},
\end{equation}
where $\epsilon>0$ and the square root and division are applied elementwise.

Consider the convex quadratic
\begin{equation}
	f(x,y)=0.1x^2+y^2.
\end{equation}
The curvature is greater along the $y$ axis. In Figure~\ref{fig:optimisers}, SGD converges more slowly than RMSProp and Adam, while AdaGrad slows early in the run.

For training example $d$, let $\bm{x}_d$ be the input row vector. Figure~\ref{fig:dnn} shows the components of the activations with the example index suppressed. For hidden layer $k$, let $W^{(k)}$ be its weight matrix, $b^{(k)}$ its bias row vector, and $f^{(k)}$ its activation function. Starting with $\bm{h}_d^{(0)}=\bm{x}_d$, compute
\begin{equation}
	\bm{u}_d^{(k)} = \bm{h}_d^{(k-1)}W^{(k)} + b^{(k)}, \qquad
	\bm{h}_d^{(k)} = f^{(k)}(\bm{u}_d^{(k)}), \qquad k=1,\dots,l.
\end{equation}
The output layer computes
\begin{equation}
	\bm{z}_d = \bm{h}_d^{(l)}U + c, \qquad
	\hat{\bm{y}}_d = g(\bm{z}_d),
\end{equation}
where $U$ is the weight matrix connecting the final hidden layer to the output layer, $c$ is the output bias row vector, and $g$ is the output activation function.

The solid arrows show how values are computed from input to loss. The dashed arrows show the order in which backpropagation uses the local derivatives. Starting from $dJ/db$, multiply by $db/da$ to obtain $dJ/da$, then multiply by $da/ds$ to obtain

    \frac{dJ}{ds}=\frac{dJ}{db}\frac{db}{da}\frac{da}{ds}.

In a network, a neuron can feed into several neurons in the next layer. The chain rule then adds the contributions through those neurons. Backpropagation reuses the resulting derivative at each layer to compute gradients for its weights and to continue backwards.

\subsubsection{Cost Function and Output Units}

The training objective is generally non-convex in the network parameters $\bm{\theta}$, so gradient descent does not guarantee convergence to a global optimum. A common objective is the mean negative log-likelihood of a probabilistic model $p_{\mathrm{model}}(\bm{y}\mid\bm{x};\bm{\theta})$, where $\bm{\theta}$ collects all the weights and biases.
\begin{equation}
	\label{eq:loss_NN_optim}
	J(\bm{\theta}) = -\E_{(\bm{x},\bm{y}) \sim \hat{p}_{\mathrm{data}}} [\log p_{\mathrm{model}}(\bm{y}\mid\bm{x};\bm{\theta})].
\end{equation}
The choice of output unit determines the appropriate likelihood and loss:
\begin{enumerate}
	\item \textbf{Linear.} For regression, $\hat{\bm{y}}_d=\bm{z}_d$. A Gaussian model $\mathcal{N}(\bm{y}_d\mid\hat{\bm{y}}_d,\bm{I})$ gives the per-example loss $J_d=\frac12\|\hat{\bm{y}}_d-\bm{y}_d\|_2^2$ up to an additive constant.
	
	\item \textbf{Sigmoid.} For binary classification, the scalar prediction $\hat{y}_d=\mathrm{sig}(z_d)$ is the Bernoulli probability $p_{\mathrm{model}}(y_d=1\mid\bm{x}_d;\bm{\theta})$.
	
	\item \textbf{Softmax.} For multiclass classification, $\hat{\bm{y}}_d=\mathrm{softmax}(\bm{z}_d)$ gives the categorical class probabilities.
	\[
	\mathrm{softmax}(\bm{z}_d)_i = \frac{e^{z_{di}}}{\sum_j e^{z_{dj}}}.
	\]
\end{enumerate}

For a training set of $N$ pairs $(\bm{x}_d,\bm{y}_d)$, the expectation under the empirical distribution $\hat{p}_{\mathrm{data}}$ in Equation~\eqref{eq:loss_NN_optim} is the sample average
\begin{equation}
	\label{eq:loss_NN_optim_emp}
	J(\bm{\theta}) = \frac{1}{N}\sum_{d=1}^N J_d, \qquad
	J_d=-\log p_{\mathrm{model}}(\bm{y}_d\mid\bm{x}_d;\bm{\theta}).
\end{equation}
The rest of this section is devoted to the calculation of this loss and the recursive calculation of its gradient, with respect to the parameters $\theta$. 

\subsubsection{Forward Propagation}

For training pair $(\bm{x}_d,\bm{y}_d)$, the forward pass computes the prediction $\hat{\bm{y}}_d$ and the per-example loss $J_d=L(\hat{\bm{y}}_d,\bm{y}_d)$. The mini-batch objective is the mean of the per-example losses.

\begin{algorithm}[H]
	\caption{Forward Propagation}
	\textbf{Require:} Training pair $(\bm{x}_d,\bm{y}_d)$, depth $l$, hidden-layer weights $W^{(k)}$, biases $b^{(k)}$, and activations $f^{(k)}$, output weights $U$, bias $c$, and activation $g$, and loss function $L$.\\
	\begin{algorithmic}[1]
		\State $\bm{h}_d^{(0)} \gets \bm{x}_d$
		\For{$k = 1,\dots,l$}
			\State $\bm{u}_d^{(k)} \gets \bm{h}_d^{(k-1)}W^{(k)} + b^{(k)}$
			\State $\bm{h}_d^{(k)} \gets f^{(k)}(\bm{u}_d^{(k)})$
		\EndFor
		\State $\bm{z}_d \gets \bm{h}_d^{(l)}U+c$
		\State $\hat{\bm{y}}_d \gets g(\bm{z}_d)$
		\State $J_d \gets L(\hat{\bm{y}}_d,\bm{y}_d)$
		\State Return $\hat{\bm{y}}_d$ and $J_d$
	\end{algorithmic}
\end{algorithm}

\subsubsection{Backward Propagation – Preliminaries}

The \textbf{backpropagation} algorithm propagates cost information backwards to compute gradients efficiently. In practice, this is achieved by means of \textbf{automatic differentiation}, which provides fast and scalable computation, thus enabling modern deep learning.

Consider a mini-batch of $N$ training pairs $(\bm{x}_d,\bm{y}_d)$. Let $J_d=L(\hat{\bm{y}}_d,\bm{y}_d)$ be the loss for example $d$, so $J=\frac{1}{N}\sum_{d=1}^N J_d$. For scalar regression with squared error, take $J_d=\frac12(\hat{y}_d-y_d)^2$.

Let $H^{(k)}$ and $A^{(k)}$ be the matrices whose rows are the hidden activations $\bm{h}_d^{(k)}$ and pre-activations $\bm{u}_d^{(k)}$, respectively, and let $H^{(0)}$ contain the inputs. Define the hidden-layer error by
\[
    \delta_{dj}^{(k)}=\frac{\partial J_d}{\partial u_{dj}^{(k)}}.
\]
Since $\partial u_{dj}^{(k)}/\partial w_{ij}^{(k)}=h_{di}^{(k-1)}$, the chain rule gives $\partial J_d/\partial w_{ij}^{(k)}=h_{di}^{(k-1)}\delta_{dj}^{(k)}$. Stacking the errors into the matrix $\Delta^{(k)}=(\delta_{dj}^{(k)})$ and averaging over the batch gives
\begin{equation}
    \frac{\partial J}{\partial W^{(k)}}=\frac{1}{N}(H^{(k-1)})^\top\Delta^{(k)}.
\end{equation}

\subsubsection{Backward Propagation – Hidden Layers}

For $k=l-1,\dots,1$, assume the hidden activations are applied elementwise. The chain rule gives
\[
    \delta_{dj}^{(k)}
    =(f^{(k)})'(u_{dj}^{(k)})\sum_a w_{ja}^{(k+1)}\delta_{da}^{(k+1)},
\]
where $a$ ranges over the neurons in layer $k+1$. In matrix form,
\begin{equation}
    \Delta^{(k)}=\bigl(\Delta^{(k+1)}(W^{(k+1)})^\top\bigr)
    \odot(f^{(k)})'(A^{(k)}),
\end{equation}
where $\odot$ denotes elementwise multiplication and the activation derivative is evaluated elementwise.

\begin{mdframed}[style=discusion, frametitle={Backpropagation through a sum}]
The hidden-layer recursion above sums one contribution from each neuron that receives the activation of neuron $j$. The diagram illustrates this chain-rule step with two paths. A scalar activation $h$ feeds into $a=F(h)$ and $b=G(h)$, whose outputs are added to form $z=a+b$. Let $J$ be the downstream loss.

\begin{center}
    \begin{tikzpicture}[stage/.style={draw, rounded corners, minimum width=9mm, minimum height=7mm}]

        \node[stage] (second) at (2.6,-0.95) {$b$};
        \node[stage] (sum) at (5.2,0) {$z=a+b$};

        \draw[->] (second.east) -- (sum.south west);
        \draw[->] (sum.east) -- (loss.west);
        \draw[->, dashed] (loss.south west) to[bend left=20] node[below] {$dJ/dz$} (sum.south east);
        \draw[->, dashed] (sum.west) -- (first.south east);
        \draw[->, dashed] (sum.west) -- (second.north east);

    \end{tikzpicture}

The solid arrows show the forward computation. The dashed arrows show the backward paths, labelled with the local factors after $dJ/dz$. Each summand changes $z$ at unit rate, so the gradient arriving at the sum is passed to both branches:
\begin{equation*}
    \frac{\partial J}{\partial a}
    =\frac{\partial J}{\partial b}
    =\frac{dJ}{dz}.
\end{equation*}
Each branch then multiplies that gradient by its own local derivative. Because both branches depend on the same $h$, their contributions add:
\begin{equation*}
    \frac{dJ}{dh}
    =\frac{\partial J}{\partial a}F'(h)
    +\frac{\partial J}{\partial b}G'(h).
\end{equation*}
In the hidden-layer recursion, $h$ plays the role of $h_{dj}^{(k)}$. Each path goes through a neuron $a$ in layer $k+1$ and contributes $w_{ja}^{(k+1)}\delta_{da}^{(k+1)}$. Their sum gives the derivative with respect to $h_{dj}^{(k)}$. Multiplying by $(f^{(k)})'(u_{dj}^{(k)})$ carries it one step further back to $\delta_{dj}^{(k)}$.
\end{mdframed}

\subsubsection{Backward Propagation – Output Layer}

Let $\bm{z}_d=\bm{h}_d^{(l)}U+c$ be the output pre-activation and let $\Delta^{\mathrm{out}}$ have rows $\partial J_d/\partial\bm{z}_d$. For scalar regression with a linear output and the squared loss above, $\hat{y}_d=z_d$, so
\begin{equation}
    \label{eq:capa_output}
    \delta_d^{\mathrm{out}}=\hat{y}_d-y_d.
\end{equation}
The final hidden-layer error is obtained by propagating the output error through $U$:
\[
    \Delta^{(l)}=\bigl(\Delta^{\mathrm{out}}U^\top\bigr)
    \odot(f^{(l)})'(A^{(l)}).
\]
The remaining parameter gradients are
\begin{equation}
    \frac{\partial J}{\partial b^{(k)}}=\frac{1}{N}\mathbf{1}^\top\Delta^{(k)},\quad
    \frac{\partial J}{\partial U}=\frac{1}{N}(H^{(l)})^\top\Delta^{\mathrm{out}},\quad
    \frac{\partial J}{\partial c}=\frac{1}{N}\mathbf{1}^\top\Delta^{\mathrm{out}},
\end{equation}
where $\mathbf{1}$ is an $N$-dimensional column vector of ones.

\begin{algorithm}[H]
    \caption{Backward Propagation}
    \textbf{Require:} Mini-batch, current parameters $\bm{\theta}$, and learning rate $\lambda$.\\
    \begin{algorithmic}[1]
        \State Run forward propagation and store the pre-activations, hidden activations, and predictions.
        \State Compute $\Delta^{\mathrm{out}}$ for the chosen output activation and loss.
        \State Compute $\partial J/\partial U$ and $\partial J/\partial c$.
        \State Compute $\Delta^{(l)}$ from $\Delta^{\mathrm{out}}$ using $U$.
        \For{$k=l,\dots,1$}
            \State Compute $\partial J/\partial W^{(k)}$ and $\partial J/\partial b^{(k)}$.
            \If{$k>1$}
                \State Compute $\Delta^{(k-1)}$ using the hidden-layer recursion.
            \EndIf
        \EndFor
        \State Update all parameters: $\bm{\theta}\gets\bm{\theta}-\lambda\nabla_{\bm{\theta}}J$.
    \end{algorithmic}
\end{algorithm}

\begin{remark}
	The algorithm is applied to each mini-batch for several \textbf{epochs}. Increasing epochs generally decreases training error, but excessive training may degrade generalisation due to \textbf{overfitting}.
\end{remark}

\subsubsection{Regularisation}

Neural networks are often large models; controlling their complexity is crucial. \textbf{Regularisation} techniques aim to reduce the generalisation error while retaining sufficient model capacity.

\paragraph{L2 Regularisation.} Adds a quadratic penalty on weights:
\[
\tilde{J}(\bm{\theta}) = J(\bm{\theta}) + \frac{\alpha}{2}
\left(\sum_{k=1}^l\|W^{(k)}\|_{\mathrm{F}}^2+\|U\|_{\mathrm{F}}^2\right).
\]
The hyperparameter $\alpha$ controls the regularisation strength, and $\|\cdot\|_{\mathrm{F}}$ is the Frobenius norm. With a gradient descent step of size $\lambda$, this penalty gives \textit{weight decay}. For example, the update for hidden-layer weights is
\[
W^{(k)} \gets (1-\beta)W^{(k)} - \lambda \frac{\partial J}{\partial W^{(k)}}, \quad \beta = \lambda \alpha.
\]
